/**
 * 注意： 
 * 1. 当前脚本需开启 PaddleOCR 功能 和 遥测功能
 * 2. 使用前需要将蓝图添加为收藏
 */

/**
 * 辅助函数：封装按键点击，减少冗余代码
 */
fn click_key(key_name, hold_time, post_sleep) {
    press(key_name);
    sleep(hold_time);
    release(key_name);
    sleep(post_sleep);
}

/**
 * 辅助函数：检查文本中是否包含数组中的所有关键字
 */
fn contains_all_keywords(text_str, keywords) {
    for kw in keywords {
        if text_str.index_of(kw) < 0 {
            return false;
        }
    }
    return true;
}

// ==================== 页面判断核心逻辑 ====================

fn check_page_match(page_type, ocr_text) {
    if page_type == "main_menu"    { return contains_all_keywords(ocr_text, ["剧情", "车辆", "地平线", "在线", "创意中心", "商店" , "世界地图"]); }
    if page_type == "competition"  { return contains_all_keywords(ocr_text, ["精选", "热门", "本月最佳", "赛事"]); }
    if page_type == "running"      { return ocr_text.index_of("安娜") >= 0; }
    if page_type == "home"         { return contains_all_keywords(ocr_text, ["剧情", "购买与出售", "车辆", "可自定义车库"]); }
    if page_type == "garage"       { return contains_all_keywords(ocr_text, ["我的车辆", "前往制造商"]); }
    if page_type == "start_run"    { return contains_all_keywords(ocr_text, ["开始","赛事", "难度与设置", "退出比赛"]); }
    if page_type == "select_run"   { return contains_all_keywords(ocr_text, ["单人", "合作", "玩家对战"]); }
    if page_type == "is_finish_run"{ return contains_all_keywords(ocr_text, ["继续", "重新开始", "观看回放影片"]); }
    return false;
}

/**
 * 动态等待页面加载（带超时机制）
 * @param page_type 页面类型字符串
 * @param timeout_sec 最大等待秒数
 */
fn wait_for_page(page_type, timeout_sec) {
    for i in 0..timeout_sec {
        let current_text = ocr(1);
        if check_page_match(page_type, current_text) {
            return true;
        }
        sleep(1000); // 每秒检测一次，降低CPU与OCR服务器压力
    }
    return false;
}

/**
 * 线性控制右扳机（油门）
 */
fn set_throttle(value, duration) {
    set_trigger("Right", value);
    sleep(duration);
}

/**
 * 从 OCR 文本中精准提取“XX技术点数可用”中的数字
 */
fn extract_skill_points(ocr_text) {
    let text_str = `${ocr_text}`;
    let anchor = "技术点数可用";
    let anchor_idx = text_str.index_of(anchor);
    
    if anchor_idx < 0 {
        return -1; // 未捕获到目标锚点
    }
    
    let walk_idx = anchor_idx - 1;
    let digits = "0123456789";
    let result_str = "";
    let found_digits = false;
    
    // 向左最多回溯 15 个字符
    for step in 0..15 {
        if walk_idx < 0 { break; }
        
        let char_item = text_str.sub_string(walk_idx, 1);
        
        if digits.index_of(char_item) >= 0 {
            result_str = char_item + result_str;
            found_digits = true;
        } else {
            // 如果已经切出数字，一旦遇到非数字阻断（如空格），立刻收网，防止把前面的其他数字误吸进来
            if found_digits { break; }
        }
        walk_idx = walk_idx - 1;
    }
    
    if result_str.len > 0 {
        return result_str.to_int();
    }
    return 0;
}

// ==================== 主程序 ====================

fn main() {
    log("Forza OCR 挂机自动化脚本 2.0 启动...");
    set_default_device(0);

    // 1. 状态初始化：退回到主菜单
    let init_retry = 0;
    while true {
        let text_str = ocr(1);
        log("当前OCR识别内容: " + text_str);
        
        if check_page_match("main_menu", text_str) {
            break;
        }
        
        log("当前非主菜单页，正在尝试按键返回...");
        if check_page_match("running", text_str) {
            click_key("START", 83, 2500); // 增加转场等待，防连打
        } else {
            click_key("B", 83, 2000);
        }
        
        init_retry = init_retry + 1;
        if init_retry > 10 {
            throw "【错误】初始化异常：无法自动退回到主菜单，请手动处理后重新运行脚本。";
        }
    }

    log("成功进入主菜单！准备计算挂机圈数...");
    
    // 2. 查看车辆并精准提取点数
    click_key("RB", 100, 500);
    let current_text = ocr(1);
    let skill = extract_skill_points(current_text);
    log("提取到当前技术点数为: " + skill);
    
    // 动态计算循环圈数 (防止上限溢出)
    let LOOP_COUNT = 100; // 默认兜底圈数
    if skill >= 0 {
        LOOP_COUNT = (999 - skill) / 10;
        if LOOP_COUNT <= 0 { LOOP_COUNT = 1; }
    } else {
        log("【警告】未能成功提取到技术点，采用默认兜底圈数：" + LOOP_COUNT);
    }
    log("🎯 最终判定挂机总轮数设置为: " + LOOP_COUNT);

    // 切换到“创意中心”标签页 (从车辆页往右切3次)
    for i in 0..3 {
        click_key("RB", 100, 500);
    }

    // 3. 进入赛事蓝图页
    sleep(1500);
    click_key("A", 83, 2000);
    click_key("A", 117, 1000);

    if !wait_for_page("competition", 10) {
        throw "进入赛事页失败，脚本终止";
    }
    log("成功进入赛事页");

    // 4. 前往【我的收藏】
    log("正在连切 7 次 RB 移动到我的收藏...");
    for i in 0..7 {
        click_key("RB", 100, 450);
    }
    sleep(2000);

    // 5. 选择蓝图并确认比赛类型
    click_key("A", 83, 3000);
    if !wait_for_page("select_run", 15) {
        throw "进入[选择比赛类型]页超时"; 
    }
    click_key("A", 83, 1000);

    // 6. 动态等待选择车辆页
    if !wait_for_page("garage", 10) {
        throw "进入[选择车辆]页超时"; 
    }
    log("成功进入车辆选择界面");

    // 7. 锁定车辆，首次进入起跑页面
    click_key("A", 83, 1500);

    // ==================== 核心刷图循环段 ====================
    log("🚀 开始执行核心挂机循环...");

    for current_loop in 1..=LOOP_COUNT {
        log("========================================");
        log("进度提示：当前正在执行第 " + current_loop + " / " + LOOP_COUNT + " 轮");
        log("========================================");

        // 等待起跑准备页加载完成（将黑屏加载时间放宽到35秒，防大作加载卡顿）
        if !wait_for_page("start_run", 35) {
             click_key("START", 83, 1000); // 尝试按菜单键拯救一下
             throw "【严重错误】进入起跑准备页超时，脚本安全退出。"; 
        }
        log("赛事加载完毕，启动赛事！");
      
        // 8. 开启比赛
        click_key("A", 83, 4500);
        
        // 9. 线性防滑起步
        set_throttle(0.0471, 16);
        set_throttle(0.2314, 16);
        set_throttle(0.4039, 17);
        set_throttle(0.5608, 17);
        set_throttle(0.7961, 16);
        set_throttle(0.8471, 67);
        
        // 10. 全油门死锁挂机阶段
        log("油门已锁死，车辆全速冲刺中...");
        let safety_counter = 0;   
        let max_drive_time = 25;  // 25次 * 2秒 = 50秒安全卡口（可根据具体蓝图耗时微调）

        while true {
            set_trigger("Right", 1.0000); // 底层强制锁死满油门
            sleep(2000); // 每2秒做一次物理及遥测状态检查
            
            safety_counter = safety_counter + 1;

            // 检查遥测：若游戏原生判断比赛已结束，直接跳出
            let tel = get_telemetry();
            if !tel.is_race_on {
                log("【遥测通知】检测到赛事已结束（已冲线）。");
                break; 
            }

            if safety_counter >= max_drive_time {
                log("【安全保护】单轮行驶时长触发熔断阈值，强行收油，避免死循环！");
                break;
            }
        }
        
        // 冲线后松开油门/减速刹车
        set_throttle(0.4353, 17);
        set_throttle(0.0941, 16);
        set_throttle(0.0000, 100);
        
        // 11. 动态检测结算页面
        if !wait_for_page("is_finish_run", 20) {
             log("【警告】未在规定时间内匹配到结算页面，尝试按START键唤出菜单保护。");
             click_key("START", 83, 1500);
             throw "异常：丢失比赛结算页"; 
        }
        log("已安全抵达比赛结算页。");

        // 最后一圈判断：完工退出
        if current_loop == LOOP_COUNT {
            log("🎉 恭喜！已完成预设的所有挂机轮数，任务完美结束！");
            break;
        }

        // 12. 自动点击【重新开始】
        log("正在执行：自动选择【重新开始】...");
        click_key("X", 83, 1500); // 呼出重新开始提示
        click_key("A", 83, 2500); // 确认重新开始并等待黑屏
    }

    // 退出整个循环后，退回到自由驾驶大地图界面
    click_key("A", 83, 2000);
    click_key("A", 83, 2000);
    log("所有自动化挂机流程已安全执行完毕，感谢使用！");
}
