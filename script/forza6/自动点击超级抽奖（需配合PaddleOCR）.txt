/**
 * 辅助函数：封装按键点击
 * @param count 连续点击次数，默认 1 次
 */
fn click_key(key_name, hold_time, post_sleep, count) {
    let loops = 1;
    if count > 0 {
        loops = count;
    }
    
    for i in 0..loops {
        press(key_name);
        sleep(hold_time);
        release(key_name);
        sleep(post_sleep);
    }
}

/**
 * 辅助函数：检查文本中是否包含数组中的所有关键字
 */
fn contains_all_keywords(text_str, keywords) {
    for kw in keywords {
        if text_str.index_of(kw) < 0 {
            return false;
        }
    }
    return true;
}

fn is_my_car_page(ocr_text) { return contains_all_keywords(`${ocr_text}`, ["我的地平线","返回住所", "我的地产"]); }
fn is_none_page(ocr_text) { return contains_all_keywords(`${ocr_text}`, ["已无剩余抽奖"]); }
fn is_none_super_page(ocr_text) { return contains_all_keywords(`${ocr_text}`, ["已无剩余超级抽奖"]); }
fn is_main_menu_page(ocr_text) {return contains_all_keywords(ocr_text, ["剧情", "车辆", "地平线", "在线", "创意中心", "商店" , "世界地图"]);}
fn is_running_page(ocr_text) {return ocr_text.index_of("安娜") >= 0;}


fn main(){
    log(" Forza OCR 挂机自动抽奖脚本启动...");
    set_default_device(0);
    //进入超级抽奖页面
    while true {
        let text_str = ocr(1);
        log("当前ocr识别内容 : " + text_str);
        
        if is_main_menu_page(text_str) {
            break;
        }
        
        log("当前页面非主菜单页，正在尝试切换...");
        if is_running_page(text_str) {
            click_key("START", 83, 2000,1);
        } else {
            click_key("B", 83, 2000,1);
        }
    }
    click_key("RB",80,1000,2);
    click_key("LEFT",80,1000,1);


    while true{
      click_key("A",80,1000,2);
      let currentOcr = ocr(1);
      let flag1 = is_my_car_page(currentOcr);
      let flag2 = is_none_page(currentOcr);
      let flag3 = is_none_super_page(currentOcr);
      if(flag1 || flag2 || flag3){
        log("抽奖结束，脚本退出");
        if(flag2 || flag3){
          click_key("A",80,1000,1);
        }
        break;
      }
    }
    

}
